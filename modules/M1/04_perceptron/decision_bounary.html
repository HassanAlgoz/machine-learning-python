Interactive University Acceptance Model

Linear Decision Boundary

The equation \(w_1(\text{Test}) + w_2(\text{Grades}) + b = 0\) separates the accepted (blue) from the rejected (red).

viewof w1 = Inputs.range([-10, 10], {value: 1, step: 0.01, label: "w₁ Test"})
viewof w2 = Inputs.range([-10, 10], {value: 1, step: 0.01, label: "w₂ Grade"})
viewof b = Inputs.range([-10, 10], {value: -1, step: 0.01, label: "b"})

data = Array.from({length: 40}, () => ({
  test: Math.random() * 10,
  grade: Math.random() * 10
}))

html`<b>Decision Rule:</b> $x_2 = -\frac{w_1}{w_2}x_1 - \frac{b}{w_2}$`
Plot.plot({
  grid: true,
  x: { domain: [0, 10], interval: 1, label: "Test (x₁)" },
  y: { domain: [0, 10], interval: 1, label: "Grade (x₂)" },
  marks: [
    Plot.ruleY([0], {strokeOpacity: 0.2}),
    Plot.ruleX([0], {strokeOpacity: 0.2}),
    Plot.dot(data, {
      x: "test",
      y: "grade",
      fill: d => (w1 * d.test + w2 * d.grade + b > 0) ? "#4e79a7" : "#e15759",
      r: 5
    }),
    Plot.line([0, 10], {
      x: x1 => x1,
      y: x1 => (-w1 * x1 - b) / w2,
      stroke: "black",
      strokeWidth: 3,
      strokeDasharray: "5,5"
    }),
    Plot.text([{x: 0.5, y: 9.5}], {
      text: d => w2 !== 0 ? "Boundary Active" : "Undefined (w₂=0)",
      frameAnchor: "top-left",
      dy: 10
    })
  ]
})
